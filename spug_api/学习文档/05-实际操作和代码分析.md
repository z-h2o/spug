### 2.1 用户登录流程分析

**步骤1：前端发送登录请求**
```javascript

    headers: {
        'Content-Type': 'application/json',
    },
    body: JSON.stringify(loginData)

```

**步骤2：后端处理登录请求**
```python
# apps/account/views.py - login函数分析
def login(request):
    # 1. 数据解析和验证
    form, error = JsonParser(
        Argument('username', help='请输入用户名'),
        Argument('password', help='请输入密码'),
        Argument('type', required=False)
    ).parse(request.body)
    
    # 2. 查找用户
    user = User.objects.filter(username=form.username, type=form.type).first()
    
    # 3. 验证密码
    if user and user.verify_password(form.password):
        # 4. 生成访问令牌
        user.access_token = uuid.uuid4().hex
        user.token_expired = time.time() + settings.TOKEN_TTL
        user.save()
        
        # 5. 返回用户信息
        return json_response({
            'access_token': user.access_token,
            'nickname': user.nickname,
            'permissions': list(user.page_perms)
        })
```

### 2.2 主机管理操作流程

**添加主机的完整流程：**

```python
# 1. 前端发送添加主机请求
POST /api/host/
{
    "name": "测试服务器",
    "hostname": "192.168.1.100",
    "port": 22,
    "username": "root",
    "password": "server_password",
    "group_ids": [1, 2],
    "desc": "用于测试的服务器"
}

# 2. 后端处理 - apps/host/views.py
class HostView(View):
    @auth('host.host.add|host.host.edit')  # 权限检查
    def post(self, request):
        # 数据验证
        form, error = JsonParser(
            Argument('name', help='请输主机名称'),
            Argument('hostname', handler=str.strip, help='请输入主机名或IP'),
            Argument('port', type=int, help='请输入SSH端口'),
            Argument('username', handler=str.strip, help='请输入登录用户名'),
            Argument('password', required=False),
            Argument('group_ids', type=list, help='请选择主机分组'),
        ).parse(request.body)
        
        if error is None:
            # 3. 验证SSH连接
            if not _do_host_verify(form):
                return json_response(error='主机连接验证失败')
            
            # 4. 创建主机记录
            if form.id:  # 更新
                host = Host.objects.get(pk=form.id)
                host.update_by_dict(form)
            else:  # 创建
                host = Host.objects.create(
                    created_by=request.user,
                    **form
                )
            
            # 5. 设置主机分组
            group_ids = form.pop('group_ids', [])
            host.groups.set(group_ids)
            
            return json_response()
```

### 3.1 需求分析：添加主机标签功能

**需求**：为主机添加标签功能，方便分类和筛选

**实现步骤：**

**步骤1：修改数据模型**
```python
# apps/host/models.py - 添加标签模型
class HostTag(models.Model, ModelMixin):
    """主机标签"""
    name = models.CharField(max_length=50, unique=True)
    color = models.CharField(max_length=7, default='#1890ff')  # 标签颜色
    desc = models.CharField(max_length=255, null=True)
    created_at = models.CharField(max_length=20, default=human_datetime)
    created_by = models.ForeignKey(User, models.PROTECT, related_name='+')
    
    class Meta:
        db_table = 'host_tags'

# 修改Host模型，添加标签关联
class Host(models.Model, ModelMixin):
    # ... 原有字段 ...
    tags = models.ManyToManyField(HostTag, blank=True, related_name='hosts')
    
    def to_view(self):
        tmp = self.to_dict()
        # 添加标签信息
        tmp['tags'] = [{'id': tag.id, 'name': tag.name, 'color': tag.color} 
                       for tag in self.tags.all()]
        return tmp
```

```python
# apps/host/views.py - 添加标签管理视图
class HostTagView(View):
    @auth('host.tag.view')
    def get(self, request):
        """获取标签列表"""
        tags = HostTag.objects.all()
        return json_response([tag.to_dict() for tag in tags])
    
    @auth('host.tag.add|host.tag.edit')
    def post(self, request):
        """创建/更新标签"""
        form, error = JsonParser(
            Argument('id', type=int, required=False),
            Argument('name', help='请输入标签名称'),
            Argument('color', default='#1890ff'),
            Argument('desc', required=False),
        ).parse(request.body)
        
        if error is None:
            if form.id:
                tag = HostTag.objects.get(pk=form.id)
                tag.update_by_dict(form)
            else:
                HostTag.objects.create(created_by=request.user, **form)
        
        return json_response(error=error)
    
    @auth('host.tag.del')
    def delete(self, request):
        """删除标签"""
        form, error = JsonParser(
            Argument('id', type=int, help='请指定要删除的标签')
        ).parse(request.GET)
        
        if error is None:
            tag = HostTag.objects.get(pk=form.id)
            if tag.hosts.exists():
                return json_response(error='该标签正在被使用，无法删除')
            tag.delete()
        
        return json_response(error=error)
```

## 📊 第五部分：性能优化实践

### 5.1 数据库查询优化

**问题：N+1查询**
```python
# 低效的代码
hosts = Host.objects.all()
for host in hosts:
    print(host.groups.all())  # 每次循环都查询数据库

# 优化后的代码
hosts = Host.objects.prefetch_related('groups').all()
for host in hosts:
    print(host.groups.all())  # 使用预加载的数据
```

**使用select_related和prefetch_related**
```python
# select_related：用于ForeignKey和OneToOneField
hosts = Host.objects.select_related('hostextend', 'created_by').all()

# prefetch_related：用于ManyToManyField和反向ForeignKey
hosts = Host.objects.prefetch_related('groups', 'tags').all()

# 组合使用
hosts = Host.objects.select_related('hostextend').prefetch_related('groups').all()
```

### 5.2 缓存优化

**Redis缓存使用**
```python
from django.core.cache import cache

# 缓存用户权限
def get_user_perms(user_id):
    cache_key = f'user_perms_{user_id}'
    perms = cache.get(cache_key)
    
    if perms is None:
        user = User.objects.get(id=user_id)
        perms = list(user.page_perms)
        cache.set(cache_key, perms, 3600)  # 缓存1小时
    
    return perms

# 缓存主机列表
def get_host_list():
    cache_key = 'host_list'
    hosts = cache.get(cache_key)
    
    if hosts is None:
        hosts = Host.objects.select_related('hostextend').all()
        hosts = [host.to_view() for host in hosts]
        cache.set(cache_key, hosts, 300)  # 缓存5分钟
    
    return hosts
```
