# consumer/consumers.py - WebSocket消费者基类
class BaseConsumer(AsyncWebsocketConsumer):
    """WebSocket基础消费者类"""
    
    async def connect(self):
        """建立WebSocket连接"""
        # 获取用户认证信息
        self.user = self.scope.get('user')
        if not self.user or not self.user.is_authenticated:
            await self.close()
            return
        
        # 加入通信组
        await self.channel_layer.group_add(
            self.group_name,
            self.channel_name
        )
        await self.accept()
    
    async def disconnect(self, close_code):
        """断开WebSocket连接"""
        await self.channel_layer.group_discard(
            self.group_name,
            self.channel_name
        )
    
    async def receive(self, text_data):
        """接收前端消息"""
        data = json.loads(text_data)
        # 处理不同类型的消息
        await self.handle_message(data)
    
    async def send_message(self, event):
        """发送消息到前端"""
        await self.send(text_data=json.dumps(event['message']))

# consumer/consumers.py - 命令执行消费者
class ExecConsumer(BaseConsumer):
    """命令执行WebSocket消费者"""
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        # 从URL获取令牌和模块信息
        token = self.scope['url_route']['kwargs']['token']
        module = self.scope['url_route']['kwargs']['module']
        
        # 根据模块设置Redis键
        if module == 'build':
            self.key = f'{settings.BUILD_KEY}:{token}'
        elif module == 'request':
            self.key = f'{settings.REQUEST_KEY}:{token}'
        elif module == 'host':
            self.key = token
        
        self.rds = get_redis_connection()
    
    def receive(self, text_data='', **kwargs):
        """处理前端消息"""
        if text_data.isdigit():
            # 获取指定索引的响应数据
            index = int(text_data)
            response = self.get_response(index)
            
            # 持续发送新数据
            while response:
                index += 1
                self.send(text_data=response)
                response = self.get_response(index)
    
    def get_response(self, index):
        """从Redis获取响应数据"""
        counter = 0
        while counter < 30:  # 最多等待6秒
            response = self.rds.lindex(self.key, index)
            if response:
                return response.decode()
            counter += 1
            time.sleep(0.2)
        return None
```

# apps/exec/executors.py - 任务执行器
class Job:
    """任务执行类"""
    
    def __init__(self, key, name, hostname, port, username, pkey, command, interpreter, params=None, token=None):
        # 创建SSH连接
        self.ssh = SSH(hostname, port, username, pkey)
        self.key = key
        self.command = self._handle_command(command, interpreter)
        self.token = token
        self.rds = get_redis_connection()
        
        # 设置环境变量
        self.env = dict(
            SPUG_HOST_ID=str(self.key),
            SPUG_HOST_NAME=name,
            SPUG_HOST_HOSTNAME=hostname,
            SPUG_SSH_PORT=str(port),
            SPUG_SSH_USERNAME=username,
            SPUG_INTERPRETER=interpreter
        )
        
        # 添加自定义参数
        if isinstance(params, dict):
            self.env.update({f'_SPUG_{k}': str(v) for k, v in params.items()})
    
    def run(self):
        """执行任务"""
        try:
            # 发送开始状态
            self.send_status(0)
            
            # 执行命令
            exit_code = self.ssh.exec_command_with_stream(
                command=self.command,
                environment=self.env,
                callback=self.send  # 实时输出回调
            )
            
            # 发送结束状态
            self.send_status(exit_code)
            
        except Exception as e:
            self.send(f'执行异常: {str(e)}')
            self.send_status(-1)
        finally:
            self.ssh.close()
    
    def send(self, data):
        """发送数据到Redis"""
        self._send({'key': self.key, 'data': data})
    
    def send_status(self, code):
        """发送状态码"""
        self._send({'key': self.key, 'status': code})
    
    def _send(self, message):
        """发布消息到Redis"""
        self.rds.publish(self.token, json.dumps(message))
```

# apps/schedule/scheduler.py - 任务调度器
from apscheduler.schedulers.blocking import BlockingScheduler
from apscheduler.triggers.cron import CronTrigger

class TaskScheduler:
    """任务调度器"""
    
    def __init__(self):
        self.scheduler = BlockingScheduler()
        self.jobs = {}
    
    def start(self):
        """启动调度器"""
        # 加载所有启用的任务
        self.load_tasks()
        
        # 启动调度器
        self.scheduler.start()
    
    def load_tasks(self):
        """加载任务"""
        from apps.schedule.models import Task
        
        for task in Task.objects.filter(is_active=True):
            try:
                # 解析Cron表达式
                trigger = CronTrigger.from_crontab(task.trigger)
                
                # 添加任务到调度器
                self.scheduler.add_job(
                    func=self.execute_task,
                    trigger=trigger,
                    args=[task.id],
                    id=str(task.id),
                    replace_existing=True
                )
                
                self.jobs[task.id] = task
                
            except Exception as e:
                print(f'加载任务失败 {task.name}: {e}')
    
    def execute_task(self, task_id):
        """执行任务"""
        task = self.jobs.get(task_id)
        if not task:
            return
        
        try:
            # 创建任务执行记录
            from apps.schedule.models import TaskHistory
            history = TaskHistory.objects.create(
                task=task,
                status='1'  # 执行中
            )
            
            # 执行任务逻辑
            if task.type == 'shell':
                self.execute_shell_task(task, history)
            elif task.type == 'python':
                self.execute_python_task(task, history)
            
        except Exception as e:
            history.status = '3'  # 失败
            history.output = str(e)
            history.save()
    
    def execute_shell_task(self, task, history):
        """执行Shell任务"""
        # 获取目标主机
        hosts = task.targets.all()
        
        for host in hosts:
            try:
                ssh = host.get_ssh()
                exit_code, output = ssh.exec_command(task.command)
                
                # 记录执行结果
                history.output += f'Host: {host.name}\\n{output}\\n'
                
                if exit_code != 0:
                    history.status = '3'  # 失败
                    
            except Exception as e:
                history.output += f'Host: {host.name} Error: {str(e)}\\n'
                history.status = '3'
        
        # 如果没有失败，标记为成功
        if history.status != '3':
            history.status = '2'  # 成功
        
        history.save()
```

# apps/monitor/executors.py - 监控执行器
class MonitorJob:
    """监控任务"""
    
    def __init__(self, detection):
        self.detection = detection
        self.rds = get_redis_connection()
    
    def run(self):
        """执行监控检查"""
        try:
            if self.detection.type == '1':  # 站点监控
                result = self.check_website()
            elif self.detection.type == '2':  # 端口监控
                result = self.check_port()
            elif self.detection.type == '3':  # 进程监控
                result = self.check_process()
            elif self.detection.type == '4':  # 自定义脚本监控
                result = self.check_script()
            
            # 处理监控结果
            self.handle_result(result)
            
        except Exception as e:
            self.handle_result({
                'is_success': False,
                'message': str(e)
            })
    
    def check_website(self):
        """检查网站可用性"""
        import requests
        
        try:
            response = requests.get(
                self.detection.addr,
                timeout=self.detection.extra.get('timeout', 30)
            )
            
            # 检查状态码
            expected_code = self.detection.extra.get('status_code', 200)
            if response.status_code == expected_code:
                return {'is_success': True, 'message': '网站正常'}
            else:
                return {
                    'is_success': False, 
                    'message': f'状态码错误: {response.status_code}'
                }
                
        except requests.RequestException as e:
            return {'is_success': False, 'message': str(e)}
    
    def check_port(self):
        """检查端口连通性"""
        import socket
        
        try:
            host, port = self.detection.addr.split(':')
            port = int(port)
            
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(10)
            result = sock.connect_ex((host, port))
            sock.close()
            
            if result == 0:
                return {'is_success': True, 'message': '端口连通正常'}
            else:
                return {'is_success': False, 'message': '端口连接失败'}
                
        except Exception as e:
            return {'is_success': False, 'message': str(e)}
    
    def handle_result(self, result):
        """处理监控结果"""
        from apps.monitor.models import DetectionHistory
        from apps.alarm.models import Alarm
        
        # 创建监控历史记录
        history = DetectionHistory.objects.create(
            detection=self.detection,
            is_success=result['is_success'],
            message=result['message']
        )
        
        # 检查是否需要报警
        if not result['is_success']:
            self.trigger_alarm(result['message'])
    
    def trigger_alarm(self, message):
        """触发报警"""
        from libs.spug import Notification
        
        # 创建通知对象
        notification = Notification(
            grp=[self.detection.group_id],
            event='1',  # 告警事件
            target=self.detection.name,
            title=f'{self.detection.name} 监控告警',
            message=message,
            duration=''
        )
        
        # 发送通知
        modes = self.detection.group.modes.split(',')
        notification.dispatch_monitor(modes)

# libs/gateway.py - API网关
class APIGateway:
    """API网关"""
    
    def __init__(self):
        self.services = {
            'user': 'http://user-service:8001',
            'host': 'http://host-service:8002',
            'exec': 'http://exec-service:8003',
            'deploy': 'http://deploy-service:8004',
        }
        self.rate_limiter = RateLimiter()
    
    def route_request(self, request):
        """路由请求到对应服务"""
        # 解析请求路径
        service_name = self.extract_service_name(request.path)
        
        if service_name not in self.services:
            return JsonResponse({'error': '服务不存在'}, status=404)
        
        # 限流检查
        if not self.rate_limiter.allow_request(request):
            return JsonResponse({'error': '请求过于频繁'}, status=429)
        
        # 负载均衡
        service_url = self.get_service_instance(service_name)
        
        # 转发请求
        return self.forward_request(request, service_url)
    
    def extract_service_name(self, path):
        """从路径提取服务名"""
        parts = path.strip('/').split('/')
        if len(parts) >= 2 and parts[0] == 'api':
            return parts[1]
        return None
    
    def get_service_instance(self, service_name):
        """获取服务实例（负载均衡）"""
        # 简单的轮询负载均衡
        return self.services[service_name]
    
    def forward_request(self, request, service_url):
        """转发请求到目标服务"""
        import requests
        
        # 构建目标URL
        target_url = f"{service_url}{request.get_full_path()}"
        
        # 转发请求
        response = requests.request(
            method=request.method,
            url=target_url,
            data=request.body,
            headers=self.get_forwarded_headers(request),
            timeout=30
        )
        
        # 返回响应
        return HttpResponse(
            content=response.content,
            status=response.status_code,
            content_type=response.headers.get('content-type')

# libs/cache.py - 缓存装饰器
def cache_result(key_prefix, timeout=3600):
    """缓存结果装饰器"""
    def decorator(func):
        def wrapper(*args, **kwargs):
            from django.core.cache import cache
            
            # 生成缓存键
            cache_key = f"{key_prefix}:{hash(str(args) + str(kwargs))}"
            
            # 尝试从缓存获取
            result = cache.get(cache_key)
            if result is not None:
                return result
            
            # 执行函数并缓存结果
            result = func(*args, **kwargs)
            cache.set(cache_key, result, timeout)
            
            return result
        return wrapper
    return decorator

# 使用示例
@cache_result('host_list', timeout=300)
def get_host_list(user_id):
    """获取用户可访问的主机列表"""
    user = User.objects.get(id=user_id)
    if user.is_supper:
        hosts = Host.objects.all()
    else:
        host_ids = get_host_perms(user)
        hosts = Host.objects.filter(id__in=host_ids)
    
    return [host.to_view() for host in hosts]
```

```python
# libs/metrics.py - 指标收集
class MetricsCollector:
    """指标收集器"""
    
    def __init__(self):
        self.metrics = {}
        self.rds = get_redis_connection()
    
    def increment_counter(self, name, value=1, tags=None):
        """递增计数器"""
        key = f"metrics:counter:{name}"
        if tags:
            key += f":{':'.join(f'{k}={v}' for k, v in tags.items())}"
        
        self.rds.incr(key, value)
    
    def record_histogram(self, name, value, tags=None):
        """记录直方图"""
        key = f"metrics:histogram:{name}"
        if tags:
            key += f":{':'.join(f'{k}={v}' for k, v in tags.items())}"
        
        # 记录到时序数据
        timestamp = int(time.time())
        self.rds.zadd(key, {f"{timestamp}:{value}": timestamp})
        
        # 保留最近1小时的数据
        cutoff = timestamp - 3600
        self.rds.zremrangebyscore(key, 0, cutoff)
    
    def record_gauge(self, name, value, tags=None):
        """记录仪表盘指标"""
        key = f"metrics:gauge:{name}"
        if tags:
            key += f":{':'.join(f'{k}={v}' for k, v in tags.items())}"
        
        self.rds.set(key, value)

# 中间件集成指标收集
class MetricsMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response
        self.metrics = MetricsCollector()
    
    def __call__(self, request):
        start_time = time.time()
        
        response = self.get_response(request)
        
        # 记录请求指标
        duration = time.time() - start_time
        self.metrics.increment_counter('http_requests_total', tags={
            'method': request.method,
            'status': response.status_code
        })
        self.metrics.record_histogram('http_request_duration_seconds', duration, tags={
            'method': request.method
        })
        
        return response
```
