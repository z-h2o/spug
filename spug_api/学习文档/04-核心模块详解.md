### 1.2 权限控制系统
```python
# apps/account/models.py - 用户权限
class User(models.Model):
    @property
    def page_perms(self):
        """页面权限"""
        data = self.get_perms_cache()  # 从缓存获取
        if data:
            return data
        
        # 遍历用户的所有角色
        for item in self.roles.all():
            if item.page_perms:
                perms = json.loads(item.page_perms)
                for m, v in perms.items():      # 模块
                    for p, d in v.items():      # 页面
                        # 生成权限字符串：模块.页面.动作
                        data.update(f'{m}.{p}.{x}' for x in d)
        
        self.set_perms_cache(data)  # 缓存权限
        return data
    
    def has_perms(self, codes):
        """检查是否有指定权限"""
        if self.is_supper:  # 超级用户拥有所有权限
            return True
        return self.page_perms.intersection(codes)
```

**权限系统设计思路**：
```
用户 → 角色 → 权限
User → Role → Permissions

权限格式：模块.页面.动作
例如：host.host.add（主机模块.主机页面.添加动作）
```

# apps/host/models.py
class Host(models.Model):
    """主机基础信息"""
    name = models.CharField(max_length=100)      # 主机名称
    hostname = models.CharField(max_length=50)   # 主机地址
    port = models.IntegerField(null=True)        # SSH端口
    username = models.CharField(max_length=50)   # SSH用户名
    pkey = models.TextField(null=True)           # SSH私钥
    
    def get_ssh(self, pkey=None, default_env=None):
        """获取SSH连接对象"""
        pkey = pkey or self.private_key
        return SSH(self.hostname, self.port, self.username, pkey, default_env=default_env)

class HostExtend(models.Model):
    """主机扩展信息"""
    host = models.OneToOneField(Host, on_delete=models.CASCADE)
    cpu = models.IntegerField()                  # CPU核数

class Group(models.Model):
    """主机分组"""
    name = models.CharField(max_length=50)
    parent_id = models.IntegerField(default=0)   # 父分组ID（支持层级）
    sort_id = models.IntegerField(default=0)     # 排序
    hosts = models.ManyToManyField(Host, related_name='groups')
    
    def to_view(self, with_hosts=False):
        """转换为前端需要的数据格式"""
        response = dict(
            key=self.id, 
            value=self.id, 
            title=self.name, 
            children=[]
        )
        if with_hosts:
            def make_item(x):
                return dict(
                    title=x.name, 
                    hostname=x.hostname, 
                    key=f'{self.id}_{x.id}', 
                    id=x.id, 
                    isLeaf=True  # 前端树形组件需要
                )
            response['children'] = [make_item(x) for x in self.hosts.all()]
        return response

## 🔧 第三部分：数据验证和处理机制

### 3.1 JsonParser 数据验证器
```python
# libs/__init__.py
class JsonParser:
    """JSON数据解析和验证器"""
    def __init__(self, *arguments):
        self.arguments = arguments
    
    def parse(self, data):
        """解析和验证数据"""
        try:
            data = json.loads(data) if isinstance(data, (str, bytes)) else data
        except json.JSONDecodeError:
            return None, '请求数据格式错误'
        
        form = AttrDict()  # 属性字典，支持点号访问
        for argument in self.arguments:
            value = data.get(argument.name)
            
            # 必填验证
            if argument.required and value is None:
                return None, argument.help or f'{argument.name}为必填项'
            
            # 类型转换
            if value is not None and argument.type:
                try:
                    value = argument.type(value)
                except (ValueError, TypeError):
                    return None, f'{argument.name}类型错误'
            
            # 自定义处理器
            if value is not None and argument.handler:
                value = argument.handler(value)
            
            # 自定义过滤器
            if value is not None and argument.filter:
                if not argument.filter(value):
                    return None, argument.help or f'{argument.name}验证失败'
            
            form[argument.name] = value if value is not None else argument.default
        
        return form, None

# 使用示例
form, error = JsonParser(
    Argument('name', help='请输入名称'),
    Argument('port', type=int, help='请输入端口号'),
    Argument('email', required=False),
    Argument('username', handler=str.strip),  # 自动去除空格
    Argument('password', filter=lambda x: len(x) >= 8, help='密码至少8位')
).parse(request.body)
```

# libs/mixins.py
def auth(perm_code):
    """权限验证装饰器"""
    def decorator(func):
        def wrapper(self, request, *args, **kwargs):
            if not request.user.has_perms([perm_code]):
                return json_response(error='无权限访问')
            return func(self, request, *args, **kwargs)
        return wrapper
    return decorator

# 使用示例

### 4.2 中间件认证机制
```python
# libs/middleware.py
class AuthenticationMiddleware:
    """认证中间件"""
    def __init__(self, get_response):
        self.get_response = get_response
    
    def __call__(self, request):
        # 检查请求路径是否需要认证
        if self._is_excluded_path(request.path):
            return self.get_response(request)
        
        # 获取访问令牌
        token = self._get_token_from_request(request)
        if not token:
            return JsonResponse({'code': 1, 'message': '未提供访问令牌'})
        
        # 验证令牌
        user = self._validate_token(token)
        if not user:
            return JsonResponse({'code': 1, 'message': '访问令牌无效'})
        
        # 设置当前用户
        request.user = user
        
        return self.get_response(request)
    
    def _get_token_from_request(self, request):
        """从请求中获取令牌"""
        # 从Header获取：Authorization: Bearer <token>
        auth_header = request.headers.get('Authorization', '')
        if auth_header.startswith('Bearer '):
            return auth_header[7:]
        
        # 从Cookie获取
        return request.COOKIES.get('access_token')
```

# libs/mixins.py
class ModelMixin:
    """模型混入类，提供通用方法"""
    
    def to_dict(self, selects=None, excludes=None):
        """转换为字典"""
        data = {}
        for field in self._meta.fields:
            if selects and field.name not in selects:
                continue
            if excludes and field.name in excludes:
                continue
            data[field.name] = getattr(self, field.name)
        return data
    
    def update_by_dict(self, data):
        """通过字典更新模型"""
        for key, value in data.items():
            if hasattr(self, key):
                setattr(self, key, value)
        self.save()

# 模型使用混入
class User(models.Model, ModelMixin):
    # 模型字段定义
    pass

# 使用示例

user_data = user.to_dict(excludes=['password_hash'])  # 排除密码字段
user.update_by_dict({'nickname': '新昵称'})             # 批量更新

### 5.2 查询优化技巧
```python
# 性能优化的查询示例
def get(self, request):
    # 使用 select_related 减少数据库查询
    hosts = Host.objects.select_related('hostextend')
    
    # 权限过滤
    if not request.user.is_supper:
        hosts = hosts.filter(id__in=get_host_perms(request.user))
    
    # 转换为字典，减少后续查询
    hosts = {x.id: x.to_view() for x in hosts}
    
    # 批量获取分组关系，避免N+1查询
    for rel in Group.hosts.through.objects.filter(host_id__in=hosts.keys()):
        hosts[rel.host_id]['group_ids'].append(rel.group_id)
    
    return json_response(list(hosts.values()))
```
